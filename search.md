<section class="page"><div class="wrap"><h1>搜索文章</h1><p class="page-intro">可搜索标题、摘要、正文、分类和标签，全部在浏览器本地完成。</p><input id="searchInput" class="search-input" type="search" placeholder="输入关键词，例如 PHP、AI、GitHub…" autocomplete="off"><div id="searchStatus" class="meta"></div><div id="searchResults" class="search-results"></div></div></section><script>(async function(){const input=document.getElementById('searchInput'),out=document.getElementById('searchResults'),status=document.getElementById('searchStatus');let data=[];try{data=await fetch('{{ '/search.json' | relative_url }}').then(r=>r.json())}catch(e){status.textContent='搜索索引加载失败，请稍后再试。';return}function render(){const q=input.value.trim().toLowerCase();out.innerHTML='';if(!q){status.textContent='共 '+data.length+' 篇文章';return}const terms=q.split(/\s+/).filter(Boolean);const hits=data.filter(p=>{const text=[p.title,p.description,p.content,p.category,(p.tags||[]).join(' ')].join(' ').toLowerCase();return terms.every(t=>text.includes(t))});status.textContent='找到 '+hits.length+' 篇文章';hits.forEach(p=>{const a=document.createElement('a');a.className='search-result';a.href=p.url;a.innerHTML='<small>'+p.date+(p.category?' · '+p.category:'')+'</small><h3>'+escapeHtml(p.title)+'</h3><p>'+escapeHtml((p.description||'').slice(0,180))+'</p>';out.appendChild(a)})}function escapeHtml(s){return String(s).replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]))}input.addEventListener('input',render);render()})();</script>
